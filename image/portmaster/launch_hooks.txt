# ReduxOS launcher hooks for PortMaster. Installed at image build as
# /opt/tools/PortMaster/reduxos/launch_hooks.txt and selected by "PortMaster.sh reduxos"
# (/roms/PORTS/PortMaster.sh). Sourced by PortMaster's utils/pm_launch.txt, which defines each hook's
# default; only what differs on ReduxOS is overridden. PortMaster runs as the unprivileged player
# (no sudo, so control.txt sets ESUDO=""); ports live in /roms/ports (-> /roms/PORTS).

pm_hook_env() {
  source "$controlfolder/control.txt"
  [ -f "${controlfolder}/mod_${CFW_NAME}.txt" ] && source "${controlfolder}/mod_${CFW_NAME}.txt"
  get_controls
  export HM_TOOLS_DIR=/opt/tools HM_PORTS_DIR=/roms/ports HM_SCRIPTS_DIR=/roms/ports
}

pm_hook_gui_env() {
  # noble keeps SDL2 under the multiarch dir, not /usr/lib
  export PYSDL2_DLL_PATH="/usr/lib/aarch64-linux-gnu"
  export SDL_VIDEODRIVER=kmsdrm
}

pm_hook_pre_launch() {
  export TERM=linux
  printf "\033c"; echo "Starting PortMaster."
}

pm_hook_post_launch() {
  printf "\033c"
}

pm_hook_frontend_refresh() {
  # new or removed ports show up in RetroArch's Ports playlist on return
  redux-scan-roms /roms >/dev/null 2>&1 || true
}
